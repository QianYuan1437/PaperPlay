cmake_minimum_required(VERSION 3.14)
project(paperplay_shell_tests LANGUAGES CXX RC)
add_executable(shell_identity_test shell_identity_test.cpp ../runner/shell_identity.cpp ../runner/Runner.rc)
target_compile_features(shell_identity_test PRIVATE cxx_std_17)
target_compile_definitions(shell_identity_test PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(shell_identity_test PRIVATE ../runner)
target_link_libraries(shell_identity_test PRIVATE shell32 propsys ole32 user32)
set_target_properties(shell_identity_test PROPERTIES OUTPUT_NAME "PaperPlay shell test")
enable_testing()
add_test(NAME shell_identity COMMAND shell_identity_test)
